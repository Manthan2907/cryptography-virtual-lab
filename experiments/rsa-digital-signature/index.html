FRESH INPUT: Har baar input box se latest text read karein const messageInput = document.getElementById('messageInput').value; if (!messageInput) { alert("Please enter a message!"); return; } // 2. DYNAMIC HASHING: Input string ka naya SHA-256 Hash create karein const encoder = new TextEncoder(); const data = encoder.encode(messageInput); const hashBuffer = await crypto.subtle.digest('SHA-256', data); // Hash buffer ko Hex string me convert karein const hashArray = Array.from(new Uint8Array(hashBuffer)); const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join(''); // 3. RSA SIGNATURE: Hash aur Private Key d se naya Signature calculate karein // Note: Agar aapke code me function name alag hai, toh use yahan update karein const signatureHex = computeRSASignature(hashHex); // 4. UI UPDATE: Screen par dynamic output render karein document.getElementById('inputDisplay').textContent = `Input: "${messageInput}"`; document.getElementById('sigDisplay').textContent = `Computed Signature S: 0x${signatureHex}`; });
